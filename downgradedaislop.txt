function assist.GetCurrentTurnPlayerName()
    local frame = assist.GetMatchFrame()
    local typeLbl = frame and frame:FindFirstChild("Type")
    if typeLbl and typeLbl:IsA("TextLabel") then
        local text = tostring(typeLbl.Text or ""):gsub("<[^>]->", ""):lower()
        local function includesName(name)
            name = tostring(name or ""):lower()
            if name == "" then return false end
            local from = 1
            while true do
                local first, last = string_find(text, name, from, true)
                if not first then return false end
                local before, after = string_sub(text, first - 1, first - 1), string_sub(text, last + 1, last + 1)
                if (first == 1 or not before:match("[%w_]")) and (last == #text or not after:match("[%w_]")) then return true end
                from = last + 1
            end
        end
        local candidates = Players:GetPlayers()
        
        for _, candidate in ipairs(candidates) do if includesName(candidate.Name) then return candidate.Name end end
        for _, candidate in ipairs(candidates) do if includesName(candidate.DisplayName) then return candidate.Name end end
        if text:match("your%s+turn") or text:match("you%s+are%s+typing") then return player.Name end
    end
    return nil
end

function assist.NormalizeTrapText(data)
    return (tostring(data or ""):gsub("([a-zA-Z]+%s*=%s*)(%d+)[kK]", function(prefix, amount)
        return prefix .. math.min(tonumber(amount) or 0, assist.MAX_TRAP_SCORE / 1000) .. "k"
    end))
end

function assist.ParseDictionary(text, shouldCancel)
    local words, wordSet = {}, {}
    local stats = {Entries = 0, Duplicates = 0, Invalid = 0}
    text = assist.NormalizePunctuation(text):gsub("\239\187\191", "")

    local function addToken(token)
        local word = token:gsub("^[%[%]{}()\"]+", ""):gsub("[%[%]{}()\"%.:!?]+$", "")
        stats.Entries = stats.Entries + 1
        if assist.IsValid(word) then
            if not wordSet[word] then
                wordSet[word] = true
                words[#words + 1] = word
            else
                stats.Duplicates = stats.Duplicates + 1
            end
        else
            stats.Invalid = stats.Invalid + 1
        end
    end

    for line in (text .. "\n"):gmatch("(.-)\n") do
        local definedWord = line:match("^%s*(.-)%s+%-%s+.*$")
        if definedWord then
            addToken(definedWord)
        else
            for token in line:gmatch("[^%s,;|]+") do addToken(token) end
        end
        if stats.Entries > 0 and stats.Entries % 5000 == 0 then
            if shouldCancel and shouldCancel() then return nil, nil, nil, "CANCELLED" end
            task.wait()
        end
    end
    if shouldCancel and shouldCancel() then return nil, nil, nil, "CANCELLED" end
    if #words == 0 then return nil, nil, stats, "No valid entries found in the dictionary" end
    return words, wordSet, stats
end

function assist.LoadExternalFiles(baseWords, baseSet, proMode, shouldCancel)
    baseWords = baseWords or assist.baseDictionary
    baseSet = baseSet or assist.baseDictionarySet
    if proMode == nil then proMode = assist.proModeEnabled end
    local data = {
        baseDictionary = baseWords, baseDictionarySet = {},
        dictionary = {}, dictionarySet = {}, TrapSuffixes = {}, TrapSuffixLengths = {}, TrapCountByLength = {[2] = 0, [3] = 0, [4] = 0},
        TrapRanks = {}, TrapSolves = {}, WordTrapCandidates = {}, TrapScores = {},
        radarPrefixIndex = {}, prefixWordIndex = {}, oneByOneWords = {}, WordTrapSuffix = {}, addedWords = {},
    }
    local prefixCounts, giverSuffixes = {}, {}
    for index, word in ipairs(baseWords) do
        if proMode or not word:find("'", 1, true) then
            for length = 2, math.min(4, #word) do
                local prefix, suffix = string_sub(word, 1, length), string_sub(word, -length)
                if #word > length then
                    prefixCounts[prefix] = (prefixCounts[prefix] or 0) + 1
                end
                if prefix ~= suffix then giverSuffixes[suffix] = true end
            end
        end
        if index % 5000 == 0 then
            if shouldCancel and shouldCancel() then return nil, "CANCELLED" end
            task.wait()
        end
    end
    local trapCount = 0
    for suffix in pairs(giverSuffixes) do
        if not baseSet[suffix] and not assist.BannedTrapSuffixes[suffix]
            and (prefixCounts[suffix] or 0) > 0 and prefixCounts[suffix] <= assist.MAX_TRAP_SOLVES
            and suffix:find("[a-z]")
            and not suffix:find("[-'][-']")
            and (proMode or not suffix:find("'", 1, true)) then
            data.TrapSuffixes[suffix] = assist.MAX_TRAP_SCORE
            data.TrapSuffixLengths[#suffix] = true
            data.TrapCountByLength[#suffix] = data.TrapCountByLength[#suffix] + 1
            trapCount = trapCount + 1
        end
    end
    for suffix in pairs(data.TrapSuffixes) do data.TrapSolves[suffix] = {} end
    for index, word in ipairs(baseWords) do
        data.dictionary[#data.dictionary + 1] = word
        data.dictionarySet[word] = true
        data.baseDictionarySet[word] = true
        if #(word:gsub("[^a-z]", "")) >= 3 and (proMode or word:match("^[a-z]+$")) then
            data.oneByOneWords[#data.oneByOneWords + 1] = word
        end
        if proMode or not word:find("'", 1, true) then
            for length = 2, math.min(4, #word - 1) do
                local prefix = string_sub(word, 1, length)
                local solves = data.TrapSolves[prefix]
                if solves then solves[#solves + 1] = word end
            end
        end
        for length = 1, math.min(2, #word) do
            local prefix = string_sub(word, 1, length)
            if not data.radarPrefixIndex[prefix] then data.radarPrefixIndex[prefix] = {} end
            table_insert(data.radarPrefixIndex[prefix], word)
        end
        for length = 1, math.min(4, #word) do
            local prefix = string_sub(word, 1, length)
            if not data.prefixWordIndex[prefix] then data.prefixWordIndex[prefix] = {} end
            table_insert(data.prefixWordIndex[prefix], word)
        end
        if index % 5000 == 0 then
            if shouldCancel and shouldCancel() then return nil, "CANCELLED" end
            task.wait()
        end
    end
    for suffix, solves in pairs(data.TrapSolves) do data.TrapRanks[suffix] = math.clamp(11 - #solves, 1, 10) end
    table.sort(data.oneByOneWords)
    if shouldCancel and shouldCancel() then return nil, "CANCELLED" end
    
    for key, value in pairs(data) do assist[key] = value end
    assist.dictionaryVersion = assist.dictionaryVersion + 1
    assist.generousCache = {}
    assist.oneByOneTargetCache = {}
    assist.oneByOneAnalysisCache = {}
    assist.oneByOneRankedCache = {}
    assist.oneByOneComputing = {}
    assist.RefreshAutomaticTrapStage()
    return trapCount, 0
end

function assist.IsProOnlyWord(word)
    word = tostring(word or "")
    return string_find(word, "'", 1, true) ~= nil
end

function assist.IsWordAllowedByMode(word)
    if not assist.proModeEnabled and word:find("'", 1, true) then return false end
    if not assist.oneByOneEnabled then return true end
    return #(word:gsub("[^a-z]", "")) >= 3 and (assist.proModeEnabled or word:match("^[a-z]+$") ~= nil)
end

function assist.NormalizePunctuation(value)
    if type(value) ~= "string" then return "" end
    local word = value:gsub("<[^>]->", ""):lower()
    local apostrophes = {"\226\128\152", "\226\128\153", "\226\128\178", "\194\180", string.char(96)}
    local hyphens = {"\226\128\144", "\226\128\145", "\226\128\146", "\226\128\147", "\226\128\148", "\226\136\146"}
    for _, mark in ipairs(apostrophes) do word = word:gsub(mark, "'") end
    for _, mark in ipairs(hyphens) do word = word:gsub(mark, "-") end
    return word
end

function assist.NormalizeWordText(value)
    return assist.NormalizePunctuation(value):gsub("%s+", ""):gsub("[^a-z'%-]", "")
end

function assist.CanonicalizeWord(value, allowUnknown)
    if type(value) ~= "string" then return nil end
    local word = assist.NormalizeWordText(value)
    if not word or word == "" or not assist.IsValid(word) then return nil end
    if not allowUnknown and not assist.IsWordAllowedByMode(word) then return nil end
    if not allowUnknown and not assist.dictionarySet[word] then return nil end
    return word
end

function assist.RegisterWordUsed(w, wasTypedByLocalPlayer, allowUnknown)
    local word = assist.CanonicalizeWord(w, false) or assist.CanonicalizeWord(w, allowUnknown == true)
    if not word then return false end
    assist.pendingWords[word] = nil
    if assist.usedWords[word] then return false end
    assist.usedWords[word] = true
    assist.usedCount = assist.usedCount + 1
    assist.InvalidateTrapState()
    assist.lastBlockedWord = word
    assist.lastBlockedSource = wasTypedByLocalPlayer and "YOU" or "OPPONENT"
    assist.lastBlockedAt = time()
    assist.HideVisibleWord(word)

    if assist.yandere_dupeBtn then
        assist.yandere_dupeBtn.Text = "  Hide Used Words: " .. (assist.hideUsed and "ON" or "OFF") .. " (" .. assist.usedCount .. ")"
    end

    if playCorrectInputFeedback then
        playCorrectInputFeedback(word)
    end

    assist.TriggerAllRefreshes()
    return true
end

function assist.IsWordSuppressed(word)
    return assist.IsWordUnavailable(word, false)
end

function assist.HideVisibleWord(word)
    if not assist.hideUsed then return end
    local function removeFrom(container)
        if not container then return end
        for _, child in ipairs(container:GetChildren()) do
            if child:GetAttribute("SatWord") == word then child:Destroy() end
        end
    end
    removeFrom(assist.resultsFrame)
    removeFrom(sidebarResults)
end

function assist.ExpirePendingWords(now)
    local changed = false
    for word, submittedAt in pairs(assist.pendingWords) do
        local inFlight = word == assist.lastSubmittedWord and (assist.isTyping or (assist.lastSubmittedAt > 0 and now - assist.lastSubmittedAt < assist.SUBMISSION_TIMEOUT + 1))
        if not inFlight and (type(submittedAt) ~= "number" or now - submittedAt >= PENDING_WORD_TIMEOUT) then
            assist.pendingWords[word] = nil
            changed = true
        end
    end
    for word, untilTime in pairs(assist.softBlockedWords) do
        if now >= untilTime then assist.softBlockedWords[word] = nil; changed = true end
    end
    if changed then assist.InvalidateTrapState() end
    if changed then assist.TriggerAllRefreshes() end
end

function assist.QueuePendingWord(value, allowUnknown)
    local word = assist.CanonicalizeWord(value, allowUnknown)
    if not word or assist.IsWordUnavailable(word, true) then return nil end
    assist.pendingWords[word] = time()
    assist.InvalidateTrapState()
    assist.HideVisibleWord(word)
    assist.TriggerAllRefreshes()
    return word
end

function assist.GetKeyCode(char)
    if type(char) == "string" and #char == 1 then
        char = char:lower()
        if assist.keyboardLayout == "QWERTZ" then if char == "z" then return Enum.KeyCode.Y end; if char == "y" then return Enum.KeyCode.Z end
        elseif assist.keyboardLayout == "AZERTY" then if char == "a" then return Enum.KeyCode.Q end; if char == "q" then return Enum.KeyCode.A end; if char == "z" then return Enum.KeyCode.W end; if char == "w" then return Enum.KeyCode.Z end end
        return Enum.KeyCode[char:upper()]
    end; return nil
end

function assist.GetVirtualKeyCode(key)
    if not key then return nil end
    local name = key.Name
    if #name == 1 and name:match("^[A-Z]$") then return string.byte(name) end
    return ({Return = 0x0D, KeypadEnter = 0x0D, Backspace = 0x08, Delete = 0x2E,
        Space = 0x20, Minus = 0xBD, Quote = 0xDE, Tab = 0x09, Escape = 0x1B})[name]
end

function assist.SendKeyCode(key)
    if not key then return false end
    if VirtualInputManager then
        local pressed = pcall(function() VirtualInputManager:SendKeyEvent(true, key, false, game) end)
        if pressed then
            task.wait(0.008)
            local released = pcall(function() VirtualInputManager:SendKeyEvent(false, key, false, game) end)
            if released then return true end
        end
    end
    local virtualKey = assist.GetVirtualKeyCode(key)
    if type(keypress) == "function" and type(keyrelease) == "function" and virtualKey then
        local pressed = pcall(keypress, virtualKey)
        if pressed then
            task.wait(0.008)
            if pcall(keyrelease, virtualKey) then return true end
        end
    end
    return false
end

function assist.SimulateKey(input)
    local key
    if typeof(input) == "EnumItem" then
        key = input
    elseif type(input) == "string" and #input == 1 then
        if input == "-" then key = Enum.KeyCode.Minus
        elseif input == "'" then key = Enum.KeyCode.Quote
        else key = assist.GetKeyCode(input) end
    end
    if key and assist.SendKeyCode(key) then return true end
    if type(input) == "string" and #input == 1 and VirtualInputManager then
        return pcall(function() VirtualInputManager:SendTextInput(input) end)
    end
    return false
end

function assist.CanContinueTyping(generation, turnVersion, requireOwnTurn)
    return assist.running and generation == assist.typingGeneration and turnVersion == assist.turnVersion
        and (not requireOwnTurn or assist.IsMyTurn())
end

function assist.TypeCharacter(character, expectedBefore, canContinue)
    if not canContinue() then return false end
    local before = assist.GetCurrentGameWord()
    local function appeared()
        local actual = assist.GetCurrentGameWord()
        return #actual > #before or actual == expectedBefore .. character
    end
    local function waitForCharacter()
        for _ = 1, 8 do
            if not canContinue() then return false end
            if appeared() then return true end
            task.wait(0.025)
        end
        return canContinue() and appeared()
    end
    if assist.SimulateKey(character) and waitForCharacter() then return true end
    if not canContinue() or not VirtualInputManager then return false end
    local sent = pcall(function() VirtualInputManager:SendTextInput(character) end)
    return sent and waitForCharacter()
end

function assist.TypeWord(wordToType, prefixTyped, requireOwnTurn)
    if assist.oneByOneEnabled then return assist.TypeOneByOneLetter(wordToType, prefixTyped, requireOwnTurn) end
    if assist.isTyping or assist.isRecovering then return false end
    if assist.lastSubmittedWord and assist.pendingWords[assist.lastSubmittedWord] then return false end
    local word = assist.CanonicalizeWord(wordToType)
    if not word or assist.IsWordUnavailable(word, true) then return false end
    assist.typingGeneration = assist.typingGeneration + 1
    local generation, turnVersion = assist.typingGeneration, assist.turnVersion
    assist.isTyping = true
    local function canContinue() return assist.CanContinueTyping(generation, turnVersion, requireOwnTurn) end
    local typedCount, displayPrefix = 0, ""
    local ok, success = xpcall(function()
        local focusedBox = UserInputService:GetFocusedTextBox()
        if focusedBox then
            pcall(function() focusedBox:ReleaseFocus(false) end)
            RunService.RenderStepped:Wait()
        end
        if not canContinue() then return false end
        prefixTyped = assist.NormalizeWordText(tostring(prefixTyped or ""))
        
        local livePrefix = assist.GetCurrentGameWord()
        if assist.savedPromptTurnVersion == assist.turnVersion and assist.IsMyTurn() then
            prefixTyped = assist.GetOriginalTurnPrompt(prefixTyped)
        elseif livePrefix ~= "" then
            prefixTyped = livePrefix
        end
        if string_sub(word, 1, #prefixTyped) ~= prefixTyped then return false end
        displayPrefix = livePrefix
        if not assist.QueuePendingWord(word) then return false end
        assist.lastSubmittedWord = word
        assist.lastSubmittedAt = 0
        assist.lastSubmittedPrefix = prefixTyped
        assist.lastSubmittedDisplayLength = #displayPrefix
        assist.lastSubmittedTypedCount = 0
        assist.lastSubmissionWasAuto = requireOwnTurn == true
        assist.lastSubmissionFeedback = assist.GetWrongAnswerReason()
        local remaining = string_sub(word, #prefixTyped + 1)
        local pauseAt = math.random(4, 7)
        for i = 1, #remaining do
            if not canContinue() then return false end
            local character = string_sub(remaining, i, i)
            if assist.auto_MakeMistakes and assist.auto_SpeedMode < 4 and math.random() < 0.04 then
                local typo
                repeat typo = string.char(math.random(97, 122)) until typo ~= character
                if not assist.SimulateKey(typo) then return false end
                task.wait(math.max(0.01, 60 / math.max(1, assist.currentCPM)))
                if not canContinue() or not assist.SimulateKey(Enum.KeyCode.Backspace) then return false end
            end
            if not assist.TypeCharacter(character, string_sub(word, 1, #prefixTyped + i - 1), canContinue) then return false end
            typedCount = typedCount + 1
            if assist.lastSubmittedWord == word then assist.lastSubmittedTypedCount = typedCount end
            if assist.humanizeEnabled then
                local delay = assist.auto_SpeedMode < 4 and (60 / math.max(1, assist.currentCPM)) or 0.075
                task.wait(delay * (0.72 + math.random() * 0.68))
                if i < #remaining and i >= pauseAt then
                    task.wait(0.10 + math.random() * 0.22)
                    pauseAt = i + math.random(4, 7)
                end
            else
                task.wait(assist.auto_SpeedMode < 4 and (60 / math.max(1, assist.currentCPM)) or 0.008)
            end
        end
        task.wait(assist.humanizeEnabled and (0.10 + math.random() * 0.22) or 0.015)
        if not canContinue() then return false end
        local observed = assist.GetCurrentGameWord()
        if #observed < #displayPrefix + #remaining then return false end
        assist.lastSubmittedAt = time()
        return assist.SimulateKey(Enum.KeyCode.Return)
    end, function(message) return tostring(message) end)
    if not ok or not success then
        if generation == assist.typingGeneration then
            
            if canContinue() then
                local readable, current = pcall(assist.GetCurrentGameWord)
                if readable and type(current) == "string" and #current >= #displayPrefix then
                    for _ = 1, math.min(typedCount, #current - #displayPrefix) do
                        if not canContinue() or not assist.SimulateKey(Enum.KeyCode.Backspace) then break end
                    end
                end
            end
            assist.pendingWords[word] = nil
            if assist.lastSubmittedWord == word then
                assist.lastSubmittedWord = nil
                assist.lastSubmittedAt = 0
                assist.lastSubmittedPrefix = ""
                assist.lastSubmittedDisplayLength = 0
                assist.lastSubmittedTypedCount = 0
                assist.lastSubmissionWasAuto = false
            end
            assist.softBlockedWords[word] = time() + 2
            assist.autoAttemptKey = ""
            assist.isTyping = false
            pcall(assist.TriggerAllRefreshes)
        end
        if not ok then warn("Raxo Assist typing error: " .. tostring(success)) end
    end
    if generation == assist.typingGeneration then assist.isTyping = false end
    return ok and success == true
end

function assist.TypeOneByOneLetter(targetWord, prefixTyped, requireOwnTurn)
    if assist.isTyping or assist.oneByOneSubmittedTurn == assist.turnVersion then return false end
    local word = assist.CanonicalizeWord(targetWord)
    if not word or #(word:gsub("[^a-z]", "")) < 3 or (not assist.proModeEnabled and not word:match("^[a-z]+$")) then return false end
    local livePrefix = assist.GetCurrentGameWord()
    local prefix = requireOwnTurn and assist.GetOriginalTurnPrompt(prefixTyped) or livePrefix
    if prefix == "" then prefix = assist.NormalizeWordText(prefixTyped or "") end
    if string_sub(word, 1, #prefix) ~= prefix or #word <= #prefix then return false end
    if requireOwnTurn and not assist.IsMyTurn() then return false end
    local focusedBox = UserInputService:GetFocusedTextBox()
    if focusedBox then pcall(function() focusedBox:ReleaseFocus(false) end); RunService.RenderStepped:Wait() end
    local letter = string_sub(word, #prefix + 1, #prefix + 1)
    assist.isTyping = true
    local sent = assist.SimulateKey(letter)
    if not sent and VirtualInputManager then
        sent = pcall(function() VirtualInputManager:SendTextInput(letter) end)
    end
    if sent then
        for _ = 1, 6 do
            if #assist.GetCurrentGameWord() > #livePrefix or not assist.IsMyTurn() then break end
            task.wait(0.025)
        end
        if assist.IsMyTurn() and #assist.GetCurrentGameWord() <= #livePrefix and VirtualInputManager then
            sent = pcall(function() VirtualInputManager:SendTextInput(letter) end)
            task.wait(0.05)
        end
        if not assist.IsMyTurn() or #assist.GetCurrentGameWord() > #livePrefix then
            assist.oneByOneSubmittedTurn = assist.turnVersion
        else
            sent = false
        end
        if sent and assist.IsMyTurn() then
            assist.SimulateKey(Enum.KeyCode.Return)
        end
    end
    assist.isTyping = false
    return sent
end

function assist.GetCurrentGameWord()
    local frame = assist.GetMatchFrame()
    local cont = frame and frame:FindFirstChild("CurrentWord")
    if not cont then return "" end

    local ld = {}
    for _, c in ipairs(cont:GetChildren()) do 
        if c:IsA("GuiObject") and c.Visible then 
            local txt = c:FindFirstChild("Letter")
            if txt and txt.TextTransparency < 1 then 
                table_insert(ld, {Txt = txt.Text, X = c.AbsolutePosition.X, Y = c.AbsolutePosition.Y, Order = c.LayoutOrder or 0}) 
            end 
        end 
    end
    
    table.sort(ld, function(a, b) 
        if a.Order ~= b.Order then return a.Order < b.Order end
        if math.abs(a.Y - b.Y) > 10 then return a.Y < b.Y end
        return a.X < b.X 
    end)

    local out = ""
    for _, d in ipairs(ld) do out = out .. tostring(d.Txt) end
    return assist.NormalizeWordText(out)
end

function assist.GetOriginalTurnPrompt(liveWord)

        and assist.IsMyTurn() then
        return assist.savedTurnPrompt
    end
    return assist.NormalizeWordText(tostring(liveWord or assist.GetCurrentGameWord()))
end

function assist.IsFeedbackVisible(item)
    if item.TextTransparency and item.TextTransparency >= 1 then return false end
    local current = item
    while current and current ~= game do
        if current:IsA("GuiObject") and not current.Visible then return false end
        if current:IsA("ScreenGui") and not current.Enabled then return false end
        current = current.Parent
    end
    return true
end

function assist.GetWrongAnswerReason()
    local frame = assist.GetMatchFrame()
    if not frame then return nil end
    local root = frame.Parent or frame
    local phrases = {
        {Text = "already used", Reason = "USED"},
        {Text = "already been used", Reason = "USED"},
        {Text = "used before", Reason = "USED"},
        {Text = "word has been used", Reason = "USED"},
        {Text = "word is already used", Reason = "USED"},
        {Text = "word was used", Reason = "USED"},
        {Text = "word is not valid", Reason = "INVALID"},
        {Text = "not valid", Reason = "INVALID"},
        {Text = "invalid", Reason = "INVALID"},
        {Text = "not a word", Reason = "INVALID"},
        {Text = "invalid word", Reason = "INVALID"},
        {Text = "wrong word", Reason = "INVALID"},
        {Text = "try another", Reason = "INVALID"},
        {Text = "try again", Reason = "INVALID"}
    }
    for _, item in ipairs(root:GetDescendants()) do
        if (item:IsA("TextLabel") or item:IsA("TextButton")) and assist.IsFeedbackVisible(item) then
            local text = tostring(item.Text or ""):lower()
            for _, phrase in ipairs(phrases) do
                if string_find(text, phrase.Text, 1, true) then return phrase.Reason end
            end
        end
    end
    return nil
end

function assist.GetSubmissionRejection()
    local reason = assist.GetWrongAnswerReason()
    if not reason then assist.lastSubmissionFeedback = nil; return nil end
    
    if reason == assist.lastSubmissionFeedback then return nil end
    return reason
end

function assist.IsWordUnavailable(word, strictUnused)
    local blockedUntil = assist.softBlockedWords[word]
    if blockedUntil and time() >= blockedUntil then assist.softBlockedWords[word] = nil; blockedUntil = nil end
    return assist.Blacklist[word] == true or assist.rejectedWords[word] == true or blockedUntil ~= nil
        or ((strictUnused or assist.hideUsed) and (assist.usedWords[word] == true or assist.pendingWords[word] ~= nil))
end

function assist.RecoverFromWrongAutoWord(reason)
    if assist.isRecovering or assist.isTyping or not assist.lastSubmittedWord or not assist.lastSubmissionWasAuto then return end
    if not assist.IsMyTurn() then return end
    local word = assist.lastSubmittedWord
    local displayLength = assist.lastSubmittedDisplayLength or 0
    local turnVersion, generation = assist.turnVersion, assist.typingGeneration
    local typedCount = assist.lastSubmittedTypedCount or 0
    assist.isRecovering = true
    if reason == "INVALID" then
        assist.rejectedWords[word] = true
    elseif reason == "USED" then
        assist.RegisterWordUsed(word, true, true)
        assist.lastBlockedSource = "SERVER"
    else
        
        assist.softBlockedWords[word] = time() + 2.5
    end
    assist.pendingWords[word] = nil
    assist.InvalidateTrapState()
    local function currentJob()
        return assist.running and generation == assist.typingGeneration and turnVersion == assist.turnVersion
            and assist.lastSubmittedWord == word and assist.IsMyTurn()
    end
    task.spawn(function()
        local ok, recoveryError = xpcall(function()
            task.wait(0.05)
            if not currentJob() then return end
            local current = assist.GetCurrentGameWord()
            if #current >= displayLength then
                for _ = 1, math.min(typedCount, #current - displayLength) do
                    if not currentJob() or not assist.SimulateKey(Enum.KeyCode.Backspace) then break end
                    task.wait(assist.humanizeEnabled and 0.075 or 0.03)
                end
            end
        end, function(message) return tostring(message) end)
        if generation == assist.typingGeneration and turnVersion == assist.turnVersion then
            if assist.lastSubmittedWord == word then
                assist.lastSubmittedWord = nil
                assist.lastSubmittedAt = 0
                assist.lastSubmittedPrefix = ""
                assist.lastSubmittedDisplayLength = 0
                assist.lastSubmittedTypedCount = 0
                assist.lastSubmissionWasAuto = false
            end
            assist.isRecovering = false
            assist.autoAttemptKey = ""
            assist.TriggerAllRefreshes()
        end
        if not ok then warn("Raxo Assist recovery error: " .. tostring(recoveryError)) end
    end)
end

end

function assist.MatchesEndingFilter(word, evilV1, evilV2, v2Text, evilV3, v3_1, v3_2, v3_3)
    if evilV1 and not (word:match("[xzqjv]$")) then return false end
    if evilV2 then local s2 = (v2Text or ""):lower():match("^%s*(.-)%s*$"); if s2 and s2 ~= "" and (#word < #s2 or string_sub(word, -#s2) ~= s2) then return false end end
    if evilV3 then
        local s3_1 = (v3_1 or ""):lower():match("^%s*(.-)%s*$"); local s3_2 = (v3_2 or ""):lower():match("^%s*(.-)%s*$"); local s3_3 = (v3_3 or ""):lower():match("^%s*(.-)%s*$")
        if (s3_1 ~= "") or (s3_2 ~= "") or (s3_3 ~= "") then
            local isMatch = false
            if s3_1 ~= "" and #word >= #s3_1 and string_sub(word, -#s3_1) == s3_1 then isMatch = true end
            if s3_2 ~= "" and #word >= #s3_2 and string_sub(word, -#s3_2) == s3_2 then isMatch = true end
            if s3_3 ~= "" and #word >= #s3_3 and string_sub(word, -#s3_3) == s3_3 then isMatch = true end
            if not isMatch then return false end
        end
    end; return true
end

function assist.makeSlider(text, p, minVal, maxVal, defaultVal, callback, trapName)
    local f = Instance.new("Frame"); f.Size = UDim2.new(1, 0, 0, 50); f.BackgroundTransparency = 1; f.ZIndex = 110; f.Parent = p
    local bg = Instance.new("Frame"); bg.Size = UDim2.new(1, 0, 1, 0); bg.BackgroundColor3 = assist.panelColor; bg.ZIndex = 110; bg.Parent = f
    Instance.new("UICorner", bg).CornerRadius = UDim.new(0, 11)
    local s = Instance.new("UIStroke", bg); s.Color = assist.strokeColor; s.Thickness = 1; s.Transparency = 0.12
    
    local lbl = Instance.new("TextLabel"); lbl.Size = UDim2.new(1, -20, 0, 20); lbl.Position = UDim2.new(0, 15, 0, 5); lbl.BackgroundTransparency = 1; lbl.Text = text; lbl.TextColor3 = Color3.fromRGB(255,255,255); lbl.Font = Enum.Font.GothamMedium; lbl.TextSize = 12; lbl.TextXAlignment = Enum.TextXAlignment.Left; lbl.ZIndex = 111; lbl.Parent = bg
    
    local valLbl = Instance.new("TextLabel"); valLbl.Size = UDim2.new(0, 50, 0, 20); valLbl.Position = UDim2.new(1, -65, 0, 5); valLbl.BackgroundTransparency = 1; valLbl.Text = tostring(defaultVal); valLbl.TextColor3 = Color3.fromRGB(200, 200, 200); valLbl.Font = Enum.Font.GothamMedium; valLbl.TextSize = 12; valLbl.TextXAlignment = Enum.TextXAlignment.Right; valLbl.ZIndex = 111; valLbl.Parent = bg

    local track = Instance.new("Frame"); track.Size = UDim2.new(1, -30, 0, 4); track.Position = UDim2.new(0, 15, 0, 35); track.BackgroundColor3 = assist.bgColor; track.BorderSizePixel = 0; track.ZIndex = 111; track.Parent = bg
    Instance.new("UICorner", track).CornerRadius = UDim.new(1, 0)
    
    local fill = Instance.new("Frame"); fill.Size = UDim2.new((defaultVal - minVal) / (maxVal - minVal), 0, 1, 0); fill.BackgroundColor3 = assist.currentAccentColor; fill.BorderSizePixel = 0; fill.ZIndex = 112; fill.Parent = track
    Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)

    local knob = Instance.new("Frame"); knob.Size = UDim2.new(0, 10, 0, 10); knob.AnchorPoint = Vector2.new(0.5, 0.5); knob.Position = UDim2.new(1, 0, 0.5, 0); knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255); knob.BorderSizePixel = 0; knob.ZIndex = 113; knob.Parent = fill
    Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)

    local dragging = false
    local function update(input)
        local pos = math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local value = math.floor(minVal + (maxVal - minVal) * pos)
        fill.Size = UDim2.new(pos, 0, 1, 0)
        valLbl.Text = tostring(value)
        callback(value, trapName)
    end

    track.InputBegan:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = true; update(input) end end)
    knob.InputBegan:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = true; update(input) end end)
    local endedConnection = assist.SafeConnect(UserInputService.InputEnded, function(input) if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end end)
    local changedConnection = assist.SafeConnect(UserInputService.InputChanged, function(input) if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then update(input) end end)
    f.Destroying:Connect(function() endedConnection:Disconnect(); changedConnection:Disconnect() end)
    
    return f
end

predictNextPrefix = function(word)
    if not word or #word < 1 then return "—" end
    if assist.oneByOneEnabled then
        local target = assist.GetOneByOneTarget(word)
        if not target then
            return assist.oneByOneComputing[assist.GetOneByOneKey(word)] and "CALCULATING..." or "NO CONTINUATION"
        end
        local analysis = assist.oneByOneAnalysisCache[assist.GetOneByOneKey(word)]
        return "NEXT " .. string_sub(target, #word + 1, #word + 1):upper() .. (analysis and analysis.Forced and " | FORCED SAFE" or "")
    end
    local function getViableSequence(startLen)
        local len = math.min(#word, startLen)
        local suffix = string_sub(word, -len)
        while #suffix > 1 do
            local matches = assist.GetWordsList(suffix, 3, 1, 1, false, false, "", false, "", "", "")
            if matches and #matches >= 3 then return suffix:upper() end
            suffix = string_sub(suffix, 2)
        end
        return suffix:upper()
    end
    local p3 = getViableSequence(3)
    local p2 = getViableSequence(2)
    local p1 = getViableSequence(1)
    if p3 == p2 and p2 == p1 then return p1 
    elseif p2 == p1 then return "L3: " .. p3 .. " | L1/2: " .. p1
    else return "L3: " .. p3 .. " | L2: " .. p2 .. " | L1: " .. p1 end
end

function assist.UpdateAutomaticSuggestion(isTurn, liveWord)
    local live = assist.NormalizeWordText(tostring(liveWord or ""))

    local isLocalTurn = isTurn == true
    if not assist.autoSuggestionEnabled or not assist.wasInMatch or not isLocalTurn or live == "" then
        assist.HideAutomaticSuggestion()
        return
    end

    if assist.autoSuggestionWord and ((not assist.oneByOneEnabled and assist.IsWordSuppressed(assist.autoSuggestionWord)) or (live ~= assist.autoSuggestionPrompt and string_sub(assist.autoSuggestionWord, 1, #live) ~= live)) then assist.HideAutomaticSuggestion() end
    if assist.autoSuggestionWord then
        if live == assist.autoSuggestionWord or assist.manualSubmittedWord == assist.autoSuggestionWord then
            assist.HideAutomaticSuggestion()
            return
        end
        if assist.autoSuggestionFrame then assist.autoSuggestionFrame.Visible = true end
        return
    end

    local promptKey = tostring(assist.lastTurnName or "") .. ":" .. live
    if assist.autoSuggestionLastKey == promptKey then return end

    local ok, matches = pcall(function()
        if assist.oneByOneEnabled then
            local target = assist.GetOneByOneTarget(live)
            return target and {target} or {}
        end
        if assist.autoTrapEnabled then
            local giver = assist.FindAutomaticTrapGiver(live)
            if giver then return {giver} end
        end
        if assist.strategyMode ~= 1 then return assist.GetStrategyWordsList(live) end
        if assist.auto_SortMode == 2 then
            return assist.GetWordsList(live, 1, assist.auto_MatchMode, 2, false, false, "", false, "", "", "", nil, true, true, true)
        end
        return assist.GetWordsList(live, 1, assist.auto_MatchMode, assist.auto_SortMode, assist.auto_EvilMode, assist.auto_EvilModeV2, assist.yandere_autoEvilInputV2.Text, assist.auto_EvilModeV3, assist.yandere_autoEvilInputV3_1.Text, assist.yandere_autoEvilInputV3_2.Text, assist.yandere_autoEvilInputV3_3.Text, nil, true, true, not assist.autoTrapEnabled)
    end)
    if ok and matches and matches[1] then
        assist.ShowAutomaticSuggestion(matches[1], live, promptKey)
    else
        assist.autoSuggestionLastKey = promptKey
        assist.HideAutomaticSuggestion()
    end
end

function assist.IsValid(w)
    if type(w) ~= "string" then return false end
    w = w:lower()
    if #w < 2 and w ~= "a" and w ~= "i" then return false end
    if w == "a" or w == "i" then return true end
    
    if not w:match("^[a-z][a-z'%-]*[a-z]'?$") then return false end
    if string_find(w, "--", 1, true) or string_find(w, "''", 1, true) or string_find(w, "-'", 1, true) or string_find(w, "'-", 1, true) then return false end
    return true
end

    assist.loadingFrame.Visible = false
    assist.dragAnchor.Visible = true

            local text, readError = assist.FetchRemoteDictionary()
            if cancelled() then return end
            if not text then error(readError or "Cannot read word dictionary", 0) end
            assist.loadingDetail.Text = "Reading " .. math.floor(#text / 1024) .. " KB..."
            local words, wordSet, stats, parseError = assist.ParseDictionary(text, cancelled)

function assist.GetTrapRank(suffix)
    local remaining = 0
    for _, solve in ipairs(assist.TrapSolves[suffix] or {}) do
        if not assist.usedWords[solve] and not assist.pendingWords[solve] then remaining = remaining + 1 end
    end
    return remaining > 0 and math.clamp(11 - remaining, 1, 10) or 0
end

function assist.IsTrapActive(suffix)
    local cached = assist.trapActiveCache[suffix]
    if cached ~= nil then return cached end
    for _, solve in ipairs(assist.TrapSolves[suffix] or {}) do
        if not assist.usedWords[solve] then
            assist.trapActiveCache[suffix] = true
            return true
        end
    end
    
    assist.trapActiveCache[suffix] = false
    return false
end

function assist.GetWordTrapInfo(word)
    local cached = assist.trapWordCache[word]
    if cached then return cached end
    local info = {Rank = 0, Score = 0, Suffix = nil, Cancelled = false}
    local length = assist.currentTrapLength
    if #word >= length then
        local suffix = string_sub(word, -length)
        if not assist.BannedTrapSuffixes[suffix] and assist.TrapSuffixes[suffix] and string_sub(word, 1, length) ~= suffix then
            info.Cancelled = true
            if assist.IsTrapActive(suffix) then
                local rank = assist.GetTrapRank(suffix)
                if rank > 0 then
                    info.Rank, info.Score, info.Suffix = rank, assist.TrapSuffixes[suffix], suffix
                    info.Cancelled = false
                end
            end
        end
    end
    assist.WordTrapSuffix[word] = info.Suffix
    assist.trapWordCache[word] = info
    return info
end

function assist.FindAutomaticTrapGiver(prompt)
    prompt = assist.NormalizeWordText(tostring(prompt or ""))
    if prompt == "" then return nil, nil, nil end
    local source = assist.radarPrefixIndex[string_sub(prompt, 1, math.min(2, #prompt))] or {}
    local bestWord, bestSuffix, bestCount
    local length = assist.currentTrapLength
    local remainingBySuffix = {}
    for _, word in ipairs(source) do
        local suffix = string_sub(word, -length)
        if string_sub(word, 1, #prompt) == prompt
            and assist.TrapSuffixes[suffix]
            and not assist.BannedTrapSuffixes[suffix]
            and string_sub(word, 1, length) ~= suffix
            and assist.IsWordAllowedByMode(word)
            and not assist.IsWordUnavailable(word, true)
            and not assist.IsSelfSolveWord(word, prompt) then
            local remaining = remainingBySuffix[suffix]
            if remaining == nil then
                remaining = 0
                for _, solve in ipairs(assist.TrapSolves[suffix] or {}) do
                    if not assist.usedWords[solve]
                        and not assist.pendingWords[solve]
                        and not assist.rejectedWords[solve] then
                        remaining = remaining + 1
                    end
                end
                remainingBySuffix[suffix] = remaining
            end
            if remaining == 1 then return word, suffix, 1 end
            if remaining > 0 and (not bestCount or remaining < bestCount
                or (remaining == bestCount and word < bestWord)) then
                bestWord, bestSuffix, bestCount = word, suffix, remaining
            end
        end
    end
    return bestWord, bestSuffix, bestCount
end

function assist.GetWeirdnessScore(word)
    word = tostring(word or ""):lower()
    local score = 0
    if #word >= 8 then score = score + 3 end
    if word:match("[qxzjv]$") then score = score + 5 end
    if word:match("[^aeiou][^aeiou][^aeiou]$") then score = score + 4 end
    if word:match("[qxzjv]") then score = score + 2 end
    if not assist.COMMON_WORDS[word] then score = score + 1 end
    return score
end

function assist.IsSelfSolveWord(word, detected)
    word = tostring(word or ""):lower()
    detected = assist.NormalizeWordText(tostring(detected or ""))

    if word == "" or detected == "" then return false end

    
    if #word >= #detected and string_sub(word, -#detected) == detected then return true end

    
    local maxBorder = math.min(6, #word - 1, math.floor(#word / 2))
    for length = 1, maxBorder do
        if string_sub(word, 1, length) == string_sub(word, -length) then return true end
    end
    return false
end

function assist.HasAvailablePrefixAnswer(prefix, playedWord)
    for _, answer in ipairs(assist.prefixWordIndex[prefix] or {}) do
        if answer ~= playedWord and not assist.IsWordUnavailable(answer, true) and not assist.IsWordSuppressed(answer) then
            return true
        end
    end
    return false
end

function assist.HasDictionaryPrefixAnswer(prefix, playedWord)
    for _, answer in ipairs(assist.prefixWordIndex[prefix] or {}) do
        if answer ~= playedWord then return true end
    end
    return false
end

function assist.GetNextPlayableSuffix(word)
    for length = math.min(4, #word), 1, -1 do
        local suffix = string_sub(word, -length)
        if assist.HasAvailablePrefixAnswer(suffix, word) then return suffix end
    end
    return nil
end

function assist.IsGenerousWord(word)
    local cached = assist.generousCache[word]
    if cached ~= nil then return cached end
    local valid = #word >= 2 and assist.HasAvailablePrefixAnswer(string_sub(word, -1), word)
    if valid then
        for length = 2, math.min(4, #word) do
            if assist.HasDictionaryPrefixAnswer(string_sub(word, -length), word) then
                valid = false
                break
            end
        end
    end
    assist.generousCache[word] = valid
    return valid
end

function assist.IsEvilWord(word)
    local suffix = assist.GetNextPlayableSuffix(word)
    if not suffix then return false end
    local hasOpponentAnswer = false
    for _, answer in ipairs(assist.prefixWordIndex[suffix] or {}) do
        if answer ~= word and not assist.IsWordUnavailable(answer, true) and not assist.IsWordSuppressed(answer) then
            hasOpponentAnswer = true
            if not assist.IsGenerousWord(answer) then return false end
        end
    end
    return hasOpponentAnswer
end

function assist.GetOneByOneTarget(prefix)
    prefix = assist.NormalizeWordText(prefix or "")
    if prefix == "" or not (assist.proModeEnabled and prefix:match("^[a-z][a-z'%-]*$") or prefix:match("^[a-z]+$")) then return nil end
    local playerCount = assist.GetOneByOnePlayerCount()
    local key = assist.GetOneByOneKey(prefix)
    local cached = assist.oneByOneTargetCache[key]
    if cached ~= nil then return cached or nil end
    if assist.oneByOneComputing[key] then return nil end
    local computation = {}
    assist.oneByOneComputing[key] = computation

    local words = assist.oneByOneWords or {}
    local left, right = 1, #words + 1
    while left < right do
        local middle = math.floor((left + right) / 2)
        if string_sub(words[middle], 1, #prefix) < prefix then left = middle + 1 else right = middle end
    end

    local function newNode(value, letter)
        return {Prefix = value, Letter = letter, Count = 0, Guarantee = {}, Chance = {}, Line = {}, Sum = {}, BestGuarantee = {}, BestChance = {}, WorstGuarantee = {}, WorstChance = {}, BestReplyCount = math.huge}
    end
    local root = newNode(prefix, "")
    root.Moves = {}
    local stack = {root}
    local function finishNode(node)
        if node.Count == 0 then
            for offset = 0, playerCount - 1 do
                node.Guarantee[offset] = offset ~= 0
                node.Chance[offset] = offset == 0 and 0 or 1
                node.Line[offset] = node.Prefix
            end
        else
            for offset = 0, playerCount - 1 do
                if offset == 0 then
                    node.Guarantee[offset] = node.BestGuarantee[offset]
                    node.Chance[offset] = node.BestChance[offset]
                else
                    node.Guarantee[offset] = node.WorstGuarantee[offset]
                    node.Chance[offset] = node.Sum[offset] / node.Count
                end
            end
        end
    end
    local function addChild(parent, child)
        parent.Count = parent.Count + 1
        if parent == root then
            root.Moves[#root.Moves + 1] = {
                Letter = child.Letter, Target = child.Line[1],
                Forced = child.Guarantee[1], Chance = child.Chance[1], Replies = child.Count,
            }
        end
        for offset = 0, playerCount - 1 do
            local nextOffset = (offset + 1) % playerCount
            local guarantee, chance, line = child.Guarantee[nextOffset], child.Chance[nextOffset], child.Line[nextOffset]
            if offset == 0 then
                local oldGuarantee, oldChance = parent.BestGuarantee[offset], parent.BestChance[offset]
                if oldGuarantee == nil or (guarantee and not oldGuarantee)
                    or (guarantee == oldGuarantee and (chance > oldChance or (chance == oldChance and child.Count < parent.BestReplyCount))) then
                    parent.BestGuarantee[offset] = guarantee
                    parent.BestChance[offset] = chance
                    parent.Line[offset] = line
                    parent.BestLetter = child.Letter
                    parent.BestReplyCount = child.Count
                end
            else
                parent.Sum[offset] = (parent.Sum[offset] or 0) + chance
                local oldGuarantee, oldChance = parent.WorstGuarantee[offset], parent.WorstChance[offset]
                if oldGuarantee == nil or (not guarantee and oldGuarantee)
                    or (guarantee == oldGuarantee and chance < oldChance) then
                    parent.WorstGuarantee[offset] = guarantee
                    parent.WorstChance[offset] = chance
                    parent.Line[offset] = line
                end
            end
        end
    end
    local function popNode()
        local child = table.remove(stack)
        finishNode(child)
        addChild(stack[#stack], child)
    end

    local previous = prefix
    local processed = 0
    local version = assist.dictionaryVersion
    local generation = assist.oneByOneGeneration
    for index = left, #words do
        local word = words[index]
        if string_sub(word, 1, #prefix) ~= prefix then break end
        local shared = #prefix
        local limit = math.min(#previous, #word)
        while shared < limit and string_sub(previous, shared + 1, shared + 1) == string_sub(word, shared + 1, shared + 1) do shared = shared + 1 end
        while #stack > shared - #prefix + 1 do popNode() end
        for length = shared + 1, #word do
            stack[#stack + 1] = newNode(string_sub(word, 1, length), string_sub(word, length, length))
        end
        previous = word
        processed = processed + 1
        if processed % 3000 == 0 then
            task.wait()
            if version ~= assist.dictionaryVersion or generation ~= assist.oneByOneGeneration or not assist.running then
                if assist.oneByOneComputing[key] == computation then assist.oneByOneComputing[key] = nil end
                return nil
            end
        end
    end
    if version ~= assist.dictionaryVersion or generation ~= assist.oneByOneGeneration or not assist.running then
        if assist.oneByOneComputing[key] == computation then assist.oneByOneComputing[key] = nil end
        return nil
    end
    while #stack > 1 do popNode() end
    finishNode(root)
    table.sort(root.Moves, function(a, b)
        if a.Forced ~= b.Forced then return a.Forced end
        if assist.strategyMode == 2 and a.Forced then
            if a.Replies ~= b.Replies then return a.Replies < b.Replies end
        elseif assist.strategyMode == 3 and a.Forced then
            if #a.Target ~= #b.Target then return #a.Target < #b.Target end
        end
        if a.Chance ~= b.Chance then return a.Chance > b.Chance end
        if a.Replies ~= b.Replies then return a.Replies < b.Replies end
        if #a.Target ~= #b.Target then return #a.Target < #b.Target end
        return a.Letter < b.Letter
    end)
    local best = root.Moves[1]
    local target = best and best.Target or nil
    assist.oneByOneTargetCache[key] = target or false
    assist.oneByOneRankedCache[key] = root.Moves
    assist.oneByOneAnalysisCache[key] = {Target = target, Forced = best and best.Forced or false, Chance = best and best.Chance or 0, NextLetter = best and best.Letter or nil, PlayerCount = playerCount}
    if assist.oneByOneComputing[key] == computation then assist.oneByOneComputing[key] = nil end
    assist.autoAttemptKey = ""
    assist.autoSuggestionLastKey = ""
    if assist.sidebarEnabled and assist.lastSidebarQuery == prefix and updateSidebar then
        updateSidebar(prefix)
    end
    return target
end

function assist.GetWordsList(detected, maxCount, matchMode, sortMode, evilV1, evilV2, v2Suffix, evilV3, v3_1, v3_2, v3_3, shouldCancel, avoidSelfSolve, strictUnused, disableTrapPriority, strategyMode)
    local limit = math.max(1, maxCount or 1)
    local matches = {}
    detected = assist.NormalizeWordText(tostring(detected or ""))

    if sortMode == 2 then
        evilV1 = false
        evilV2 = false
        evilV3 = false
        v2Suffix = ""
        v3_1, v3_2, v3_3 = "", "", ""
    end

    local function isBetter(a, b)
        local scoreA = getDynTrapScore(a)
        local scoreB = getDynTrapScore(b)
        local proA = assist.IsProOnlyWord(a)
        local proB = assist.IsProOnlyWord(b)

        if not disableTrapPriority then
            local rankA, rankB = assist.GetWordTrapInfo(a).Rank, assist.GetWordTrapInfo(b).Rank
            if rankA ~= rankB then return rankA > rankB end
        end
        if not assist.proModeEnabled and proA ~= proB then return not proA end

        if sortMode == 2 then
            if #a ~= #b then return #a > #b end
            if scoreA ~= scoreB then return scoreA > scoreB end
            local weirdA, weirdB = assist.GetWeirdnessScore(a), assist.GetWeirdnessScore(b)
            if weirdA ~= weirdB then return weirdA > weirdB end
            return a < b
        end
        
        local maxTrapA = scoreA >= (assist.MAX_TRAP_SCORE * 0.99)
        local maxTrapB = scoreB >= (assist.MAX_TRAP_SCORE * 0.99)
        if not disableTrapPriority and maxTrapA ~= maxTrapB then return maxTrapA end
        if scoreA == 0 and scoreB == 0 then
            local weirdA, weirdB = assist.GetWeirdnessScore(a), assist.GetWeirdnessScore(b)
            if weirdA ~= weirdB then return weirdA > weirdB end
        end

        if sortMode == 3 then
            local sA, sB = assist.GetKillerScore(a), assist.GetKillerScore(b)
            if sA ~= sB then return sA > sB end
            if scoreA ~= scoreB then return scoreA > scoreB end
            if #a ~= #b then return #a > #b end
            return a < b
        end

        if scoreA ~= scoreB then return scoreA > scoreB end
        local sA, sB = assist.GetSmartScore(a), assist.GetSmartScore(b)
        if sA ~= sB then return sA > sB end
        if #a ~= #b then return #a > #b end
        return a < b
    end

    local function addMatch(target, word)
        if #target == limit and not isBetter(word, target[#target]) then return end
        local position = #target + 1
        for i, existing in ipairs(target) do
            if isBetter(word, existing) then position = i; break end
        end
        table_insert(target, position, word)
        if #target > limit then table.remove(target) end
    end

    local source = assist.dictionary
    if matchMode == 1 and detected ~= "" and assist.radarPrefixIndex then
        source = assist.radarPrefixIndex[string_sub(detected, 1, math.min(2, #detected))] or {}
    end

    local lastYieldTime = os_clock()
    
    for i, word in ipairs(source) do
        if os_clock() - lastYieldTime > 0.015 then
            if shouldCancel and shouldCancel() then return nil, true end
            task.wait()
            lastYieldTime = os_clock()
        end

        local strictlyBlocked = strictUnused and assist.IsWordUnavailable(word, true)
        if assist.IsWordAllowedByMode(word) and not strictlyBlocked and not assist.IsWordSuppressed(word) then
            local isValid = true
            if detected ~= "" then
                if matchMode == 1 then isValid = string_sub(word, 1, #detected) == detected
                elseif matchMode == 2 then isValid = string_find(word, detected, 1, true) ~= nil
                elseif matchMode == 3 then isValid = string_sub(word, -#detected) == detected end
            end
            
            if strictUnused and avoidSelfSolve and detected ~= "" and string_sub(word, 1, #detected) ~= detected then isValid = false end
            if isValid then isValid = assist.MatchesEndingFilter(word, evilV1, evilV2, v2Suffix or "", evilV3, v3_1 or "", v3_2 or "", v3_3 or "") end
            if isValid and strategyMode == 2 then isValid = assist.IsGenerousWord(word) end
            if isValid and strategyMode == 3 then isValid = assist.IsEvilWord(word) end
            if isValid then
                if not avoidSelfSolve or not assist.IsSelfSolveWord(word, detected) then
                    addMatch(matches, word)
                end
            end
        end
    end

    if shouldCancel and shouldCancel() then return nil, true end
    
    return matches
end

function assist.FindSpamWord(detected, suffix)
    detected = assist.NormalizeWordText(tostring(detected or ""))
    suffix = assist.NormalizeWordText(tostring(suffix or ""))
    if detected == "" or suffix == "" then return nil end

    local source = assist.dictionary
    if assist.auto_MatchMode == 1 and assist.radarPrefixIndex then
        source = assist.radarPrefixIndex[string_sub(detected, 1, math.min(2, #detected))] or {}
    end

    local best = nil
    local bestRank = nil
    for _, word in ipairs(source) do
        if assist.IsWordAllowedByMode(word) and not assist.IsWordSuppressed(word) and not assist.usedWords[word] and not assist.pendingWords[word] and not assist.rejectedWords[word] then
            local valid = true
            if assist.auto_MatchMode == 1 then valid = string_sub(word, 1, #detected) == detected
            elseif assist.auto_MatchMode == 2 then valid = string_find(word, detected, 1, true) ~= nil
            elseif assist.auto_MatchMode == 3 then valid = string_sub(word, -#detected) == detected end

            if valid and string_sub(word, -#suffix) == suffix then
                valid = assist.MatchesEndingFilter(word, assist.auto_EvilMode, assist.auto_EvilModeV2, assist.yandere_autoEvilInputV2.Text, assist.auto_EvilModeV3, assist.yandere_autoEvilInputV3_1.Text, assist.yandere_autoEvilInputV3_2.Text, assist.yandere_autoEvilInputV3_3.Text)
            end
            if valid and string_sub(word, 1, #detected) == detected and string_sub(word, -#suffix) == suffix and not assist.IsSelfSolveWord(word, detected) then
                local rank
                if assist.auto_SortMode == 2 then rank = #word * 100000 + assist.GetSmartScore(word)
                elseif assist.auto_SortMode == 3 then rank = assist.GetKillerScore(word) * 100000 + #word
                else rank = assist.GetSmartScore(word) * 100000 - #word end
                if bestRank == nil or rank > bestRank then best, bestRank = word, rank end
            end
        end
    end
    return best
end

function assist.GetSpamModeCandidate(detected)
    local count = #assist.spamSuffixes
    if count == 0 then return nil, nil end
    for offset = 0, count - 1 do
        local idx = ((assist.spamSuffixIndex - 1 + offset) % count) + 1
        local suffix = assist.spamSuffixes[idx]
        local word = assist.FindSpamWord(detected, suffix)
        if word then
            assist.spamSuffixIndex = idx
            assist.UpdateSpamStatus()
            return word, suffix
        end
    end
    return nil, nil
end

function assist.FilterSpamMatches(matches)
    if not matches or #matches == 0 then return matches, nil end
    local count = #assist.spamSuffixes
    for offset = 0, count - 1 do
        local idx = ((assist.spamSuffixIndex - 1 + offset) % count) + 1
        local suffix = assist.spamSuffixes[idx]
        local bucket = {}
        for _, word in ipairs(matches) do
            if #word >= #suffix and string_sub(word, -#suffix) == suffix then table_insert(bucket, word) end
        end
        if #bucket > 0 then
            assist.spamSuffixIndex = idx
            assist.UpdateSpamStatus()
            return bucket, suffix
        end
    end

    
    local traps = {}
    for _, word in ipairs(matches) do if getDynTrapScore(word) > 0 then table_insert(traps, word) end end
    if #traps > 0 then return traps, nil end
    return matches, nil
end

function assist.RenderRadarMatches(matches, query)
    assist.ClearRadarRows()
    assist.radarStatus.Visible = #matches == 0
    assist.radarStatus.Text = (#matches == 0) and (query == "" and "Waiting for letters..."
        or (assist.oneByOneEnabled and assist.oneByOneComputing[assist.GetOneByOneKey(query)] and "CALCULATING BEST MOVE..." or "NO SUGGESTIONS FOUND")) or ""

    for index, w in ipairs(matches) do
        local f = Instance.new("Frame")
        f.Name = "RadarRow"
        f:SetAttribute("SatWord", w)
        f.Size = UDim2.new(1, 0, 0, 40)
        f.BackgroundTransparency = 1
        f.LayoutOrder = index + 1
        f.ZIndex = 122
        f.Parent = sidebarResults

        local b = Instance.new("TextButton")
        b.Size = UDim2.new(1, 0, 1, 0)
        b.BackgroundColor3 = assist.panelColor
        b.BorderSizePixel = 0
        b.AutoButtonColor = false
        b.Text = ""
        b.ZIndex = 123
        b.Parent = f
        Instance.new("UICorner", b).CornerRadius = UDim.new(0, 6)
        local bs = Instance.new("UIStroke", b); bs.Color = assist.strokeColor; bs.Thickness = 1
        assist.BindHover(b, assist.panelColor, assist.hoverColor)

        local txt = Instance.new("TextLabel")
        txt.Size = UDim2.new(1, -15, 1, 0)
        txt.Position = UDim2.new(0, 12, 0, 0)
        txt.BackgroundTransparency = 1
        txt.RichText = true
        local trapInfo = assist.GetWordTrapInfo(w)
        local trapSuffix = trapInfo.Suffix
        local weirdWord = assist.GetWeirdnessScore(w) >= 5
        local trapScore = trapInfo.Score
        local strongTrap = trapSuffix and #trapSuffix >= 3 and #trapSuffix <= 7 and trapScore >= (assist.MAX_TRAP_SCORE * 0.9)
        local prefixText = assist.addedWords[w] and "+ " or ""
        local coloredWord = prefixText
        for letterIndex = 1, #w do
            local letter = string_sub(w, letterIndex, letterIndex)
            local color = "#FFFFFF"
            if strongTrap and letterIndex > (#w - #trapSuffix) then
                color = "#FF4D5A"
            elseif (trapSuffix or weirdWord) and letterIndex > math.max(0, #w - 3) then
                color = "#FFD166"
            end
            coloredWord = coloredWord .. '<font color="' .. color .. '">' .. letter .. '</font>'
        end
        if trapInfo.Rank > 0 then coloredWord = coloredWord .. ' <font color="#8D6CFF">[R' .. trapInfo.Rank .. '/10]</font>' end
        if assist.oneByOneEnabled then
            local nextLetter = string_sub(w, #query + 1, #query + 1):upper()
            local ranked = assist.oneByOneRankedCache[assist.GetOneByOneKey(query)] or {}
            local move = ranked[index]
            coloredWord = '<font color="#FFD166">#' .. index .. ' ' .. nextLetter .. '</font>  →  ' .. w:upper()
                .. (move and move.Forced and ' <font color="#76E0A0">[SAFE]</font>' or '')
        end
        txt.Text = coloredWord
        txt.TextSize = 22
        txt.Font = Enum.Font.GothamBold
        txt.TextXAlignment = Enum.TextXAlignment.Left
        txt.TextTruncate = Enum.TextTruncate.AtEnd
        txt.TextColor3 = Color3.fromRGB(255, 255, 255)
        txt.ZIndex = 124
        txt.Parent = b

        b.MouseButton1Click:Connect(function()
            assist.playPop()
            task.spawn(function() assist.TypeWord(w, query) end)
        end)
    end
end

updateSidebar = function(query)
    query = assist.NormalizeWordText(tostring(query or ""))

    assist.radarRequestedQuery = query
    assist.radarUpdateVersion = assist.radarUpdateVersion + 1

    if query == "" then
        assist.ClearRadarRows()
        assist.radarStatus.Text = "Waiting for letters..."
        assist.radarStatus.Visible = true
        return
    end

    assist.radarStatus.Text = "Finding words..."
    assist.radarStatus.Visible = true
    if assist.radarWorkerRunning then return end
    assist.radarWorkerRunning = true

    task.spawn(function()
        while assist.running do
            local currentVersion = assist.radarUpdateVersion
            local currentQuery = assist.radarRequestedQuery
            local v2Suffix = assist.sidebar_EvilModeV2 and assist.sbEBox.Text or ""
            local ok, matches = pcall(function()
                if assist.oneByOneEnabled then
                    return assist.GetOneByOneRankedWords(currentQuery, 8)
                end
                return assist.GetWordsList(currentQuery, 30, 1, assist.auto_SortMode, false, assist.sidebar_EvilModeV2, v2Suffix, false, "", "", "", function()
                    return currentVersion ~= assist.radarUpdateVersion or not assist.sidebarEnabled
                end, false, false, false, assist.strategyMode)
            end)

            if not ok then
                if currentVersion == assist.radarUpdateVersion then
                    assist.ClearRadarRows()
                    assist.radarStatus.Text = "Word search failed"
                    assist.radarStatus.Visible = true
                end
                assist.radarWorkerRunning = false
                return
            end

            if matches and currentVersion == assist.radarUpdateVersion and assist.sidebarEnabled then
                if assist.spamRadarEnabled and not assist.oneByOneEnabled then matches = select(1, assist.FilterSpamMatches(matches)) end
                assist.RenderRadarMatches(matches, currentQuery)
            end

            if currentVersion == assist.radarUpdateVersion then
                assist.radarWorkerRunning = false
                return
            end
        end
    end)
end

function assist.FinishTrackedTurn()
    assist.turnVersion = assist.turnVersion + 1
    assist.savedTurnPrompt = ""
    assist.savedPromptTurnVersion = -1
    if assist.oneByOneEnabled then
        assist.lastSubmittedWord = nil
        assist.manualSubmittedWord = nil
        assist.autoAttemptKey = ""
        assist.HideAutomaticSuggestion()
        local completedSolve = assist.oneByOneSubmittedTurn == assist.turnVersion - 1
        assist.oneByOneSubmittedTurn = nil
        return completedSolve
    end
    local reason = assist.GetSubmissionRejection()
    local word
    if assist.lastSubmittedWord and assist.lastSubmittedAt > 0 then
        word = assist.lastSubmittedWord
    else
        word = assist.manualSubmittedWord
        if not word or word == assist.turnPrompt then
            word = assist.lastWordInTurn
            if #word <= #assist.turnPrompt or string_sub(word, 1, #assist.turnPrompt) ~= assist.turnPrompt then
                word = assist.lastCompleteWordInTurn
            end
            if #word <= #assist.turnPrompt or string_sub(word, 1, #assist.turnPrompt) ~= assist.turnPrompt then word = nil end
        end
    end
    local completedSolve = word ~= nil and reason == nil and not assist.rejectedWords[word]
    if word and (not assist.rejectedWords[word] or reason == "USED") then
        if reason == "USED" or not reason then
            local wasMe = assist.lastTurnName == player.Name or assist.lastTurnName == player.DisplayName
            assist.RegisterWordUsed(word, wasMe, true)
        else
            assist.pendingWords[word] = nil
        end
    end
    
    assist.lastSubmittedWord = nil
    assist.lastSubmittedAt = 0
    assist.lastSubmittedPrefix = ""
    assist.lastSubmittedDisplayLength = 0
    assist.lastSubmittedTypedCount = 0
    assist.lastSubmissionWasAuto = false
    assist.lastSubmissionFeedback = nil
    assist.manualSubmittedWord = nil
    assist.autoSuggestionLastKey = ""
    assist.autoAttemptKey = ""
    assist.isAutoPlayScheduled = false
    assist.autoScheduleVersion = assist.autoScheduleVersion + 1
    assist.isRecovering = false
    assist.lastWordInTurn = ""
    assist.lastCompleteWordInTurn = ""
    assist.longestWordInTurn = ""
    assist.turnPrompt = ""
    assist.HideAutomaticSuggestion()
    return completedSolve
end

local function samePromptLetters(first, second)
    if #first ~= #second then return false end
    local counts = {}
    for i = 1, #first do
        local character = string_sub(first, i, i)
        counts[character] = (counts[character] or 0) + 1
    end
    for i = 1, #second do
        local character = string_sub(second, i, i)
        counts[character] = (counts[character] or 0) - 1
        if counts[character] < 0 then return false end
    end
    return true
end

function assist.TrackGameTurn(matchVisible, currentTurn, liveWord)
    if matchVisible then
        assist.matchEndTimer = nil
        if not assist.wasInMatch then assist.OnMatchStart() end
        if currentTurn and currentTurn ~= assist.lastTurnName then
            if assist.lastTurnName then
                local wasLocalTurn = assist.lastTurnName == player.Name
                local previousWord = assist.lastCompleteWordInTurn
                if not assist.dictionarySet[previousWord] then previousWord = assist.lastWordInTurn end
                assist.previousTurnWord = assist.dictionarySet[previousWord] and previousWord or ""
                local completedSolve = assist.FinishTrackedTurn()
                if wasLocalTurn and currentTurn ~= player.Name and completedSolve then
                    assist.turnarounds = assist.turnarounds + 1
                    local nextLength = math.min(4, 2 + math.floor(assist.turnarounds / 5))
                    if nextLength ~= assist.currentTrapLength then
                        assist.currentTrapLength = nextLength
                        assist.RefreshAutomaticTrapStage()
                    elseif assist.RenderTrapSliders then
                        assist.RenderTrapSliders()
                    end
                end
            end
            assist.lastTurnName = currentTurn
        end
        if currentTurn and liveWord ~= "" then
            if assist.turnPrompt == "" then assist.turnPrompt = liveWord end
            if currentTurn == player.Name and assist.savedTurnPrompt == ""
                and (assist.oneByOneEnabled or #liveWord <= 4) then
                local original = liveWord
                local previousWord = assist.previousTurnWord
                if not assist.oneByOneEnabled and #previousWord > #liveWord then
                    local suffix = string_sub(previousWord, -#liveWord)
                    if samePromptLetters(suffix, liveWord) then original = suffix end
                end
                assist.savedTurnPrompt = original
                assist.savedPromptTurnVersion = assist.turnVersion
            end
            assist.lastSeenGameWord = liveWord
            assist.lastWordInTurn = liveWord
            if #liveWord > #assist.longestWordInTurn then assist.longestWordInTurn = liveWord end
            if #liveWord > #assist.turnPrompt and assist.dictionarySet[liveWord] and string_sub(liveWord, 1, #assist.turnPrompt) == assist.turnPrompt then
                assist.lastCompleteWordInTurn = liveWord
            end
        end
        if assist.manualSubmittedWord and not assist.isTyping and not assist.isRecovering then
            local reason = assist.GetSubmissionRejection()
            if reason then
                local submitted = assist.manualSubmittedWord
                assist.pendingWords[submitted] = nil
                if reason == "USED" then
                    assist.RegisterWordUsed(submitted, true, true)
                    assist.lastBlockedSource = "SERVER"
                else
                    assist.rejectedWords[submitted] = true
                    assist.TriggerAllRefreshes()
                end
                assist.manualSubmittedWord = nil
                if assist.lastCompleteWordInTurn == submitted then assist.lastCompleteWordInTurn = "" end
            end
        end
    elseif assist.wasInMatch then
        assist.matchEndTimer = assist.matchEndTimer or time()
        if time() - assist.matchEndTimer >= MATCH_END_GRACE then assist.OnMatchEnd() end
    end
end
